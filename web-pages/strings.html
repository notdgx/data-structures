    char str[10],ch;
    int s = sizeof(str)/sizeof(str[0]) , i;
    printf("Enter the string -> ") ;
    for(i = 0 ; i < s-1;i++){
        ch =  getchar();
        if (ch == '\n'){
            break;
        }
        str[i] = ch;
    }
    str[i] ='\0';

    for(i = 0; i < s-1 ; i++){
        if (str[i] >= 97 && str[i] <= 122 ){
            str[i] = str[i] -32;
        }
    }

    printf("The String you entered in UPPERCASE -> %s",str);
    
    return 0;
}

void permute(char arr[], int start, int n) {
    if (start == n) {
        for (int i = 0; i < n; i++) {
            printf("%c ", arr[i]);
        }
        printf("\n");
        return;
    }

    for (int i = start; i < n; i++) {
        // swap 
        char temp = arr[start];
        arr[start] = arr[i];
        arr[i] = temp;

        permute(arr, start + 1, n);

        // undo it 
        temp = arr[start];
        arr[start] = arr[i];
        arr[i] = temp;
    }
}

void mergearr(char arr[], int u, int l)
{
    int mid = (u + l ) / 2;
    char newarr[u - l + 1];
    int i = l;
    int k = 0;
    int j = mid + 1;

    while (i <= mid && j <= u) {
        if (arr[i] <= arr[j]) {
            newarr[k] = arr[i];
            i++;
        }
        else {
            newarr[k] = arr[j];
            j++;
        }
        k++;
    }

    while (i <= mid) {
        newarr[k] = arr[i];
        i++;
        k++;
    }

    while (j <= u) {
        newarr[k] = arr[j];
        j++;
        k++;
    }

    for (int x = 0; x < k; x++) {
        arr[l + x] = newarr[x];
    }

}


void mergesort(char arr[], int l, int u){
    if (l >= u){
        return;
    }

    int mid = (l + u) / 2;

    mergesort(arr, l, mid);
    mergesort(arr, mid + 1, u);

    mergearr(arr, u, l);
}
