// Searching Algorithms -> Q 01 : Enter an integer array and an integer value from the user, then perform linear search using iteration.

#include <stdio.h>
#define SIZE 100

int main(){

    int arr[SIZE], n, target, index = -1;

    printf("Enter the size of the array: ");
    scanf("%d", &n);

    if (n <= 0 || n > SIZE){
        printf("Invalid array size\n");
        return 1;
    }

    for (int i = 0; i < n; i++){
        printf("Enter element number %d -> ", i+1);
        scanf("%d", &arr[i]);
    }

    printf("\nArray -> [");
    for (int i = 0 ; i <n ; i ++){
        printf("%d, ",arr[i]);
    }
    printf("]\n");

    printf("Enter the value to search -> ");
    scanf("%d", &target);

    for (int i = 0; i < n; i++){
        if (arr[i] == target){
            index = i;
            break;
        }
    }

    if (index != -1){
        printf("Element found at index -> %d\n", index);}
    else{
        printf("Element not found\n");}

    return 0;
}

// Searching Algorithms -> Q 05 : Enter an integer array and an integer value from the user, then perform binary search using iteration.

#include <stdio.h>
#define SIZE 100

int binary_search(int arr[], int n, int target)
{
    int low = 0, high = n - 1, mid;

    while (low <= high){
        mid = low + (high - low) / 2;

        if (arr[mid] == target) {return mid;}

        else if (arr[mid] < target) {low = mid + 1;}

        else{  high = mid - 1;}
    }
    return -1;
}

int main(){

    int arr[SIZE], n, target, index = -1;

    printf("Enter the size of the array: ");

        return 1;
    }
    
    printf("Enter the SORTED array -> \n");
    for (int i = 0; i < n; i++){
        printf("Enter element number %d -> ", i+1);
        scanf("%d", &arr[i]);
    }

    printf("\nArray -> [");
    for (int i = 0 ; i <n ; i ++){
        printf("%d, ",arr[i]);
    }
    printf("]\n");

    printf("Enter the value to search -> ");
    scanf("%d", &target);

    if (index != -1){
        printf("Element found at index -> %d\n", index);}

9. Calculate the best-case and worst-case time complexity of linear search and measure its execution time on an integer array.

Answer 9:

// Searching Algorithms -> Q 09 : Calculate the best-case and worst-case time complexity of linear search and measure its execution time on an integer array.

#include <stdio.h>
#define SIZE 100

int main(){

    int arr[SIZE], n, target, index = -1;

    int steps = 0;

    printf("Enter the size of the array: ");
    scanf("%d", &n);

    if (n <= 0 || n > SIZE){
        printf("Invalid array size\n");
        return 1;
    }

    for (int i = 0; i < n; i++){
        printf("Enter element number %d -> ", i+1);
        scanf("%d", &arr[i]);
    }

    printf("\nArray -> [");
    for (int i = 0 ; i <n ; i ++){
        printf("%d, ",arr[i]);
    }
    printf("]\n");

    printf("Enter the value to search -> ");
    scanf("%d", &target);

    for (int i = 0; i < n; i++){
        steps++;
        if (arr[i] == target){
            index = i;
            break;
        }
    }

    if (index != -1){
        printf("Element found at index -> %d\n", index);}
    else{
        printf("Element not found\n");}


    printf("Steps taken -> %d\n", steps);

    printf("\nTime Complexity ->\n");
    printf("Best case -> O(1) -> 1 step\n");
    printf("Worst case -> O(n) -> up to %d steps\n", n);

    return 0;
}

10. Calculate the best-case and worst-case time complexity of binary search and measure its execution time on an integer array.

Answer 10:

// Searching Algorithms -> Q 10 : Calculate the best-case and worst-case time complexity of binary search and measure its execution time on an integer array.

#include <stdio.h>
#define SIZE 100

int binary_search(int arr[], int n, int target, int * steps)
{
    int low = 0, high = n - 1, mid;

    while (low <= high){
        mid = low + (high - low) / 2;
        (*steps)++;

        if (arr[mid] == target) {return mid;}

        else if (arr[mid] < target) {low = mid + 1;}

        else{  high = mid - 1;}
    }
    return -1;
}

int main(){

    int arr[SIZE], n, target, index = -1;
    int steps = 0;

    printf("Enter the size of the array: ");
    scanf("%d", &n);
    
    if (n <= 0 || n > SIZE){
        printf("Invalid array size\n");
        return 1;
    }
    
    printf("Enter the SORTED array -> \n");
    for (int i = 0; i < n; i++){
        printf("Enter element number %d -> ", i+1);
        scanf("%d", &arr[i]);
    }

    printf("\nArray -> [");
    for (int i = 0 ; i <n ; i ++){
        printf("%d, ",arr[i]);
    }
    printf("]\n");

    printf("Enter the value to search -> ");
    scanf("%d", &target);

    index = binary_search(arr,n,target,&steps);

    if (index != -1){
        printf("Element found at index -> %d\n", index);}
    else{
        printf("Element not found\n");}

    
    printf("Steps taken in Binary Search -> %d\n", steps);

    printf("\nTime Complexity ->\n");
    printf("Best case -> O(1)\n");
    printf("Worst case -> O(log(n))\n");


    return 0;
}
