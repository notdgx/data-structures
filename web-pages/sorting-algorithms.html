// Sorting Algorithms -> Q 01 : Enter an integer array from the user and perform selection sort using iteration. Also calculate the best and worst time complexity with exact time.

#include <stdio.h>
#include <time.h>

#define SIZE 100

void selection_sort(int arr[], int n, long long *comparisons){

    int min, temp;

    for (int i = 0; i < n - 1; i++){
        
        min = i;
        for (int j = i + 1; j < n; j++){
            (*comparisons)++;
            if (arr[j] < arr[min]){
                min = j;
            }
        }

        if (min != i){
            temp = arr[i];
            arr[i] = arr[min];
            arr[min] = temp;
        }
    }
}

#include <stdio.h>
#include <time.h>

#define SIZE 100

void insertion_sort(int arr[], int n, long long *comparisons){

    int key, j;
    for (int i = 1; i < n; i++){
        key = arr[i];
        j = i - 1;

        while (j >= 0){
            (*comparisons)++;
            if (arr[j] > key){
                arr[j + 1] = arr[j];
                j--;}
            
            else{break;}
        }
        arr[j + 1] = key;
    }
}

// Sorting Algorithms -> Q 07 : Enter an integer array from the user and perform merge sort using iteration. Also calculate the best and worst time complexity with exact time.

#include <stdio.h>
#include <time.h>

#define SIZE 100

void merge_sort(int arr[], int n, long long *comparisons){
    int temp[SIZE];

    for (int width = 1; width < n; width *= 2){
        for (int left = 0; left < n - 1; left += 2 * width){
            int mid = left + width - 1;
            int right = left + 2 * width - 1;

            if (mid >= n - 1){continue;}

            if (right >= n){
                right = n - 1;}

            int i = left;
            int j = mid + 1;
            int k = left;

            while (i <= mid && j <= right){
                (*comparisons)++;
                if (arr[i] <= arr[j]){
                    temp[k++] = arr[i++];
                }
                else{
                    temp[k++] = arr[j++];
                }
            }

            while (i <= mid){ temp[k++] = arr[i++]; }
            while (j <= right){ temp[k++] = arr[j++]; }

            for (int p = left; p <= right; p++){
                arr[p] = temp[p];
            }
        }
    }
}

}

// Sorting Algorithms -> Q 09 : Enter an integer array from the user and perform quick sort using iteration. Also calculate the best and worst time complexity with exact time.


#include <stdio.h>
#include <time.h>

#define SIZE 100

int partition(int arr[], int low, int high, long long *comparisons){
    int pivot = arr[high];
    int i = low - 1;
    int temp;

    for (int j = low; j < high; j++){
        (*comparisons)++;

        if (arr[j] <= pivot){
            i++;

            temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }

void quick_sort(int arr[], int n, long long *comparisons){

    int stack[SIZE];
    int top = -1;
    int low, high, pivot;

    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0){
        high = stack[top--];
        low = stack[top--];

        pivot = partition(arr, low, high, comparisons);

        if (pivot - 1 > low){
            stack[++top] = low;
            stack[++top] = pivot - 1;
        }

        if (pivot + 1 < high){
            stack[++top] = pivot + 1;
            stack[++top] = high;
        }
    }
}

        for (int left = 0; left < n - 1; left += 2 * width){

            int right = left + 2 * width - 1;

            if (mid >= n - 1){continue;}

            if (right >= n){
                right = n - 1;}

            int i = left;
            int j = mid + 1;
            int k = left;

            while (i <= mid && j <= right){
                (*comparisons)++;
                if (arr[i] <= arr[j]){

                }

            while (j <= right){ temp[k++] = arr[j++]; }

            for (int p = left; p <= right; p++){

        }
    }
}

void merge(char arr[], char temp[], int left,int mid, int right,long long *comparisons){

    int i = left;
    int j = mid + 1;
    int k = left;

    while (i <= mid && j <= right){
        (*comparisons)++;
        if (arr[i] <= arr[j]){
            temp[k++] = arr[i++];
        }
        else{
            temp[k++] = arr[j++];
        }
    }

    while (i <= mid){temp[k++] = arr[i++]; }

    while (j <= right){temp[k++] = arr[j++]; }

    for (int p = left; p <= right; p++){
        arr[p] = temp[p];
    }
}

void merge_sort(char arr[], char temp[], int left, int right,long long *comparisons){

    if (left >= right){ return;}

    int mid = left + (right - left) / 2;

    merge_sort(arr, temp, left, mid, comparisons);
    merge_sort(arr, temp, mid + 1, right, comparisons);

    merge(arr, temp, left, mid, right, comparisons);
}

// Sorting Algorithms -> Q 19 : Enter a string array from the user and perform quick sort using iteration. Also calculate the best and worst time complexity with exact time.

#include <stdio.h>
#include <time.h>

#define SIZE 100

int partition(char arr[], int low, int high, long long *comparisons){
    char pivot = arr[high];
    int i = low - 1;
    char temp;

    for (int j = low; j < high; j++){
        (*comparisons)++;

        if (arr[j] <= pivot){
            i++;

            temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }

    temp = arr[i + 1];
    arr[i + 1] = arr[high];
    arr[high] = temp;

    return i + 1;
}

void quick_sort(char arr[], int n, long long *comparisons){

    int stack[SIZE];
    int top = -1;
    int low, high, pivot;

    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0){
        high = stack[top--];
        low = stack[top--];

        pivot = partition(arr, low, high, comparisons);

        if (pivot - 1 > low){
            stack[++top] = low;
            stack[++top] = pivot - 1;
        }

        if (pivot + 1 < high){
            stack[++top] = pivot + 1;
            stack[++top] = high;
        }
    }
}
